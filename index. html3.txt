  function lsGet(key, fallback) {
    try {

  }
  function lsSet(key, val) {

  //  Здесь — Web Crypto SHA-256 + соль, только для демонстрации.
  // ================================================================
  async function hashPassword(password, salt) {
    const data = new TextEncoder().encode(salt + '::' + password);
    const buf = await crypto.subtle.digest('SHA-256', data);
    return Array.from(new Uint8Array(buf)).map(b => b.toString(16).padStart(2, '0')).join('');
  }

  function genSalt() {

  //  ГЕНЕРАЦИЯ 2FA: TOTP-подобный код
  //  В реальном проекте — RFC 6238 (HMAC-SHA1, 30 сек).

  // ================================================================
  const TOTP_PERIOD = 30; // секунд
  const TOTP_WINDOW = 1;  // разрешаем ±1 окно (для неточных часов)

  function generateSecret() {
    // Base32-подобная строка из 32 символов (удобно вводить вручную)
    const alphabet = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ234567';
    const arr = new Uint8Array(32);
    crypto.getRandomValues(arr);
    return Array.from(arr).map(b => alphabet[b % 32]).join('');
  }

  async function totpCode(secret, counter) {
    // Эмуляция TOTP: SHA-256(secret + counter) → 6 цифр
    const data = new TextEncoder().encode(secret + ':' + counter);
    const buf = await crypto.subtle.digest('SHA-256', data);
    const bytes = new Uint8Array(buf);
    // Берём 4 байта и превращаем в число
    const num = ((bytes[0] & 0x7f) << 24) | (bytes[1] << 16) | (bytes[2] << 8) | bytes[3];
    return String(num % 1000000).padStart(6, '0');
  }

  function currentCounter() {
    return Math.floor(Date.now() / 1000 / TOTP_PERIOD);
  }

  async function expectedTotp(secret) {
    return totpCode(secret, currentCounter());
  }

  async function verifyTotp(secret, input) {
    const clean = String(input).replace(/\D/g, '');
    if (clean.length !== CODE_LENGTH) return false;
    const now = currentCounter();
    for (let d = -TOTP_WINDOW; d <= TOTP_WINDOW; d++) {
      const code = await totpCode(secret, now + d);
      if (code === clean) return true;
    }
    return false;
  }

  async function doLogin() {
    const ident = document.getElementById('loginIdent').value.trim().toLowerCase();
    const pass = document.getElementById('loginPassword').value;
    const err = document.getElementById('authError');
    err.textContent = '';

    if (!ident || !pass) { err.textContent = 'Заполните оба поля'; return; }

    const user = findUser(ident);
    if (!user) { err.textContent = 'Пользователь не найден'; return; }

    const hash = await hashPassword(pass, user.salt);
    if (hash !== user.passwordHash) { err.textContent = 'Неверный пароль'; return; }

    // Если у пользователя включена 2FA — запрашиваем код
    if (user.twoFA && user.totpSecret) {
      authState.pendingUserId = user.id;
      authState.mode = 'totp';
      renderAuthModal();
      return;
    }

    finishLogin(user);
  }

  function finishLogin(user) {
    setSession({ userId: user.id, loginAt: Date.now() });
    closeAuth();
    renderUserArea();
    render();
    showToast(`Добро пожаловать, ${user.name}!`);
  }

  async function doRegister() {
    const name = document.getElementById('regName').value.trim();
    const login = document.getElementById('regLogin').value.trim().toLowerCase();
    const email = document.getElementById('regEmail').value.trim().toLowerCase();
    const pass = document.getElementById('regPassword').value;
    const pass2 = document.getElementById('regPassword2').value;
    const err = document.getElementById('authError');
    err.textContent = '';

    if (!name) { err.textContent = 'Введите имя'; return; }
    if (!/^[a-z0-9_]{3,20}$/.test(login)) {
      err.textContent = 'Логин: 3–20 символов, латиница/цифры/_'; return;
    }
    if (findUser(login)) { err.textContent = 'Такой логин уже занят'; return; }
    if (email && findUser(email)) { err.textContent = 'Этот email уже используется'; return; }
    if (pass.length < 6) { err.textContent = 'Пароль минимум 6 символов'; return; }
    if (pass !== pass2) { err.textContent = 'Пароли не совпадают'; return; }

    const salt = genSalt();
    const passwordHash = await hashPassword(pass, salt);

    const users = loadUsers();
    const id = 'u_' + Date.now() + '_' + Math.random().toString(36).slice(2, 8);
    const newUser = {
      id, name, login, email: email || null,
      salt, passwordHash,
      twoFA: false, totpSecret: null,
      createdAt: Date.now()
    };
    users.push(newUser);
    if (!saveUsers(users)) { err.textContent = 'Не удалось сохранить (хранилище переполнено)'; return; }

    // Сразу предлагаем настроить 2FA
    authState.pendingUserId = id;
    authState.mode = 'setup2fa';
    authState.setupSecret = generateSecret();
    renderAuthModal();
  }

  async function confirmSetup2FA() {
    const code = getCodeFromInputs('setupCodeInputs');
    const err = document.getElementById('authError');
    err.textContent = '';

    if (code.length !== CODE_LENGTH) { err.textContent = 'Введите 6 цифр'; return; }

    const ok = await verifyTotp(authState.setupSecret, code);
    if (!ok) { err.textContent = 'Неверный код. Проверьте часы и попробуйте снова.'; return; }

    const users = loadUsers();
    const u = users.find(x => x.id === authState.pendingUserId);
    if (!u) { err.textContent = 'Пользователь не найден'; return; }
    u.twoFA = true;
    u.totpSecret = authState.setupSecret;
    saveUsers(users);

    finishLogin(u);
    showToast('Двухфакторная защита включена ✅');
  }

  function skipSetup2FA() {
    const user = loadUsers().find(x => x.id === authState.pendingUserId);
    if (!user) { closeAuth(); return; }
    finishLogin(user);
    showToast('2FA не включена. Можно включить позже в профиле.');
  }

  function renderCodeInputs(containerId) {
    const box = document.getElementById(containerId);
    if (!box) return;
    box.innerHTML = Array.from({ length: CODE_LENGTH }, (_, i) =>
      `<input type="text" inputmode="numeric" maxlength="1" data-idx="${i}" aria-label="Цифра ${i+1}">`
    ).join('');
    const inputs = [...box.querySelectorAll('input')];
    inputs.forEach((inp, i) => {
      inp.addEventListener('input', e => {
        const v = e.target.value.replace(/\D/g, '');
        e.target.value = v.slice(-1);
        if (v && i < inputs.length - 1) inputs[i + 1].focus();
      });
      inp.addEventListener('keydown', e => {
        if (e.key === 'Backspace' && !e.target.value && i > 0) inputs[i - 1].focus();
        if (e.key === 'ArrowLeft' && i > 0) inputs[i - 1].focus();
        if (e.key === 'ArrowRight' && i < inputs.length - 1) inputs[i + 1].focus();
      });
      inp.addEventListener('paste', e => {
        e.preventDefault();
        const text = (e.clipboardData || window.clipboardData).getData('text').replace(/\D/g, '');
        if (!text) return;
        for (let j = 0; j < inputs.length; j++) {
          inputs[j].value = text[j] || '';
        }
        const last = Math.min(text.length, inputs.length) - 1;
        if (last >= 0) inputs[last].focus();
      });
    });
    if (inputs[0]) inputs[0].focus();
  }

  function getCodeFromInputs(containerId) {
    const box = document.getElementById(containerId);
    if (!box) return '';
    return [...box.querySelectorAll('input')].map(i => i.value || '').join('');
  }

  function sortAds(list) {
    const arr = [...list];
    switch (currentSort) {
      case 'old': return arr.sort((a, b) => a.id - b.id);
      case 'priceAsc': return arr.sort((a, b) => a.price - b.price);
      case 'priceDesc': return arr.sort((a, b) => b.price - a.price);
      default: return arr.sort((a, b) => b.id - a.id);
    }
  }

    const me = currentUser();

        || a.desc.toLowerCase().includes(q)
        || a.city.toLowerCase().includes(q);

    document.getElementById('resultsInfo').textContent =
      sorted.length ? `Найдено объявлений: ${sorted.length}` : '';
    document.getElementById('sortSelect').value = currentSort;

    if (!sorted.length) {
      grid.innerHTML = `<div class="empty">😕 Ничего не найдено.</div>`;
      return;
    }

      const media = a.photo
        ? `<img src="${a.photo}" alt="${escapeHtml(a.title)}" loading="lazy">`
        : (emojiByCat[a.cat] || '📦');
      const mine = me && a.ownerId === me.id;

        ? `<button class="card-edit" title="Редактировать" onclick="event.stopPropagation(); openModal(${a.id})">✎</button>`
        : '';
      const delBtn = mine
        ? `<button class="card-delete" title="Удалить" onclick="event.stopPropagation(); removeAd(${a.id})">✕</button>`

  function handleFile(file) {
    if (!file.type.startsWith('image/')) { showToast('Это не изображение'); return; }
    if (file.size > MAX_FILE_MB * 1024 * 1024) { showToast(`Файл больше ${MAX_FILE_MB} МБ`); return; }
    const reader = new FileReader();
    reader.onload = ev => {
      const img = new Image();
      img.onload = () => {
        const dataUrl = compressImage(img);
        if (!dataUrl) { showToast('Не удалось обработать фото'); return; }
        currentPhotoData = dataUrl;
        previewImg.src = dataUrl;
        preview.classList.add('visible');
        dropZone.style.display = 'none';
      };
      img.onerror = () => showToast('Битый файл');
      img.src = ev.target.result;
    };
    reader.onerror = () => showToast('Не удалось прочитать файл');
    reader.readAsDataURL(file);
  }

  function compressImage(img) {
    let { width, height } = img;
    if (width > height && width > MAX_IMG_SIZE) {
      height = Math.round(height * MAX_IMG_SIZE / width); width = MAX_IMG_SIZE;
    } else if (height >= width && height > MAX_IMG_SIZE) {
      width = Math.round(width * MAX_IMG_SIZE / height); height = MAX_IMG_SIZE;
    }
    const canvas = document.createElement('canvas');
    canvas.width = width; canvas.height = height;
    const ctx = canvas.getContext('2d');
    ctx.fillStyle = '#fff'; ctx.fillRect(0, 0, width, height);
    ctx.drawImage(img, 0, 0, width, height);
    try { return canvas.toDataURL('image/jpeg', JPEG_QUALITY); }
    catch { return null; }
  }

  }

  function submitAd() {
    const me = currentUser();
    if (!me) { openAuth('login'); return; }

    const title = document.getElementById('inTitle').value.trim();
    const price = parseInt(document.getElementById('inPrice').value, 10);
    const cat = document.getElementById('inCat').value;
    const city = document.getElementById('inCity').value.trim() || 'Не указан';
    const desc = document.getElementById('inDesc').value.trim();

    if (!title) { showToast('Введите название'); return; }
    if (!price || price <= 0) { showToast('Введите корректную цену'); return; }

    const prev = ads;
    if (editingId === null) {
      ads = [{ id: nextId++, title, price, cat, city, desc, photo: currentPhotoData, ownerId: me.id }, ...ads];
    } else {
      const old = ads.find(a => a.id === editingId);
      if (!old || old.ownerId !== me.id) { showToast('Нет доступа'); return; }
      ads = ads.map(a => a.id === editingId
        ? { ...a, title, price, cat, city, desc, photo: currentPhotoData }
        : a);
    }
    if (!saveAds(ads)) { ads = prev; if (editingId === null) nextId--; return; }

    const wasEdit = editingId !== null;
    closeModal(); render();
    showToast(wasEdit ? 'Обновлено ✏️' : 'Опубликовано ✅');
  }
